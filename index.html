/** * Squeak Easy — scores worker. * * KV layout * --------- * SCORES namespace: * key: "place:{placeId}" value: JSON array of individual scores [3, 5, 4, …] * key: "summary" value: JSON { [placeId]: { avg, count } } — cached rollup * * RATE_LIMIT namespace: * key: "ip:{ip}:{placeId}" value: "1" TTL 86 400 s (one vote per place per day per IP) * * Endpoints * --------- * GET /scores → { [placeId]: { avg, count } } * POST /scores → { avg, count } (body: { placeId, score, turnstileToken }) * OPTIONS /scores → CORS preflight * * Everything else → 404. */ // --- helpers ---------------------------------------------------------------- const VALID_SCORES = [1, 2, 3, 4, 5]; const ONE_DAY = 86_400; // seconds const SUMMARY_TTL = 300; // seconds — rebuild the summary at most every 5 min const GLOBAL_RATE_LIMIT = 60; // max POSTs per IP per hour (across all places) function cors(env) { return { 'Access-Control-Allow-Origin': env.ALLOWED_ORIGIN || '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Access-Control-Max-Age': '86400', }; } function json(body, status, env) { return new Response(JSON.stringify(body), { status, headers: { 'Content-Type': 'application/json', ...cors(env) }, }); } // --- Turnstile verification ------------------------------------------------- async function verifyTurnstile(token, ip, env) { if (!env.TURNSTILE_SECRET) return true; // no secret configured → skip check if (!token) return false; const form = new URLSearchParams(); form.set('secret', env.TURNSTILE_SECRET); form.set('response', token); form.set('remoteip', ip); const res = await fetch('https://challenges.cloudflare.com/turnstile/v0/siteverify', { method: 'POST', body: form, }); const result = await res.json(); return result.success === true; } // --- rate limiting ---------------------------------------------------------- async function isRateLimited(ip, placeId, env) { // Per-place: one vote per IP per day const placeKey = `ip:${ip}:${placeId}`; const existing = await env.RATE_LIMIT.get(placeKey); if (existing) return { limited: true, reason: 'already_rated' }; // Global: cap total POSTs per IP per hour const globalKey = `rate:${ip}`; const countRaw = await env.RATE_LIMIT.get(globalKey); const count = countRaw ? parseInt(countRaw, 10) : 0; if (count >= GLOBAL_RATE_LIMIT) return { limited: true, reason: 'too_fast' }; return { limited: false }; } async function recordRateLimit(ip, placeId, env) { // Mark this IP+place combo for 24 hours await env.RATE_LIMIT.put(`ip:${ip}:${placeId}`, '1', { expirationTtl: ONE_DAY }); // Bump the hourly counter const globalKey = `rate:${ip}`; const countRaw = await env.RATE_LIMIT.get(globalKey); const count = countRaw ? parseInt(countRaw, 10) : 0; await env.RATE_LIMIT.put(globalKey, String(count + 1), { expirationTtl: 3600 }); } // --- score storage ---------------------------------------------------------- async function getPlaceScores(placeId, env) { const raw = await env.SCORES.get(`place:${placeId}`); return raw ? JSON.parse(raw) : []; } async function addScore(placeId, score, env) { const scores = await getPlaceScores(placeId, env); scores.push(score); await env.SCORES.put(`place:${placeId}`, JSON.stringify(scores)); const count = scores.length; const avg = scores.reduce((sum, s) => sum + s, 0) / count; // Invalidate the cached summary so the next GET rebuilds it. await env.SCORES.delete('summary'); return { avg, count }; } // --- summary cache ---------------------------------------------------------- async function getSummary(env) { // Try the cache first. const cached = await env.SCORES.get('summary'); if (cached) return JSON.parse(cached); // Rebuild by scanning all place:* keys. const summary = {}; let cursor = undefined; do { const list = await env.SCORES.list({ prefix: 'place:', cursor }); for (const key of list.keys) { const placeId = key.name.slice(6); // strip "place:" const scores = JSON.parse(await env.SCORES.get(key.name) || '[]'); if (scores.length > 0) { summary[placeId] = { avg: scores.reduce((s, v) => s + v, 0) / scores.length, count: scores.length, }; } } cursor = list.list_complete ? undefined : list.cursor; } while (cursor); // Cache for 5 minutes. await env.SCORES.put('summary', JSON.stringify(summary), { expirationTtl: SUMMARY_TTL }); return summary; } // --- request handler -------------------------------------------------------- export default { async fetch(request, env) { const url = new URL(request.url); // Only /scores is a valid path. if (url.pathname !== '/scores') { return json({ error: 'not found' }, 404, env); } // CORS preflight if (request.method === 'OPTIONS') { return new Response(null, { status: 204, headers: cors(env) }); } // GET /scores if (request.method === 'GET') { const summary = await getSummary(env); return json(summary, 200, env); } // POST /scores if (request.method === 'POST') { let body; try { body = await request.json(); } catch { return json({ error: 'bad json' }, 400, env); } const { placeId, score, turnstileToken } = body || {}; // Validate placeId if (typeof placeId !== 'string' || placeId.length === 0 || placeId.length > 100) { return json({ error: 'invalid placeId' }, 400, env); } // Validate score const scoreNum = typeof score === 'string' ? parseInt(score, 10) : score; if (!VALID_SCORES.includes(scoreNum)) { return json({ error: 'score must be 1–5' }, 400, env); } const ip = request.headers.get('CF-Connecting-IP') || '0.0.0.0'; // Turnstile const tokenOk = await verifyTurnstile(turnstileToken || '', ip, env); if (!tokenOk) { return json({ error: 'bot check failed' }, 403, env); } // Rate limit const limit = await isRateLimited(ip, placeId, env); if (limit.limited) { const status = limit.reason === 'already_rated' ? 409 : 429; return json({ error: limit.reason }, status, env); } // Store const result = await addScore(placeId, scoreNum, env); // Record rate limit AFTER successful write await recordRateLimit(ip, placeId, env); return json(result, 200, env); } return json({ error: 'method not allowed' }, 405, env); }, };
